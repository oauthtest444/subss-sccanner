function (event) {
          if (event.data.type === 'set_content') {
            var styleElement = window.document.createElement('style')
            styleElement.innerText = event.data.content

            var previewStyleId = 'simplesat-preview-style-tag'
            styleElement.id = previewStyleId

            var simplesatCustomStyleTag = document.getElementById(previewStyleId)
            if (simplesatCustomStyleTag) {
              document.head.replaceChild(styleElement, simplesatCustomStyleTag)
            } else {
              document.head.appendChild(styleElement)
            }
          }
        }
